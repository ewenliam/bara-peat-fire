## Context

A webhook message of type `audio` or `image` carries `{"id", "mime_type", "sha256", "url"}`.
Download is two GETs, both with header `Authorization: Bearer <token>`:

1. `GET {graph_url}/{media_id}` returns JSON
   `{"messaging_product", "id", "url", "mime_type", "sha256", "file_size"}`.
2. `GET <that url>` returns the file bytes. The URL expires after 5 minutes; a 404 means it
   expired, and the fix is to repeat step 1 once and retry.

Checked on real messages on 2026-10-29 (not stated in Meta's docs):
- The webhook `sha256` is **base64** of the SHA-256 digest (44 characters); the step-1
  `sha256` is **hex** (64 characters). Both describe the same bytes.
- `mime_type` can differ: the webhook said `audio/ogg; codecs=opus`, step 1 said `audio/ogg`.
- `file_size` is an integer.

## Interface

```python
class MediaError(Exception): ...

MAX_BYTES: int  # 16 * 1024 * 1024 (Meta's audio limit; images are capped lower by Meta)
EXTENSIONS: dict[str, str]  # base mime type -> file extension

def verify_sha256(data: bytes, expected: str) -> bool
def media_path(media_dir: Path, media_id: str, mime_type: str) -> Path
def download_media(media_id: str, token: str, media_dir: Path, expected_sha256: str | None = None,
                   session=None, graph_url: str = "https://graph.facebook.com") -> dict
```

- `EXTENSIONS`: `audio/ogg` .ogg, `audio/mpeg` .mp3, `audio/mp4` .m4a, `audio/aac` .aac,
  `audio/amr` .amr, `image/jpeg` .jpg, `image/png` .png.
- `verify_sha256`: True if `expected` is the SHA-256 of `data` as 64 hex digits (either
  case) or as standard base64 of the 32-byte digest. Compare with `hmac.compare_digest`.
  False (never raises) for anything else, including empty or malformed strings.
- `media_path`: `media_dir / f"{media_id}{ext}"`, where `ext` comes from `EXTENSIONS` using
  the mime type before any `;` (stripped, lower case). Raise `MediaError` if `media_id` is
  not a non-empty string of ASCII digits (this blocks path tricks like `../x`) or if the mime
  type is not in `EXTENSIONS`.
- `download_media`:
  - `session` defaults to `requests.Session()`. Every request is exactly
    `session.get(url, headers={"Authorization": f"Bearer {token}"}, timeout=30)`.
  - Validate `media_id` before any request (same rule as `media_path`).
  - Step 1; a non-200 status raises `MediaError` with the status code.
  - Build the path with `media_path(media_dir, media_id, meta["mime_type"])`. If that file
    already exists and `verify_sha256(existing_bytes, meta["sha256"])` is True, return
    without downloading (`"cached": True`).
  - Step 2. On 404, repeat step 1 once and retry step 2 once with the new URL. Any other
    non-200, or a second failure, raises `MediaError` with the status code.
  - Raise `MediaError` if the body is larger than `MAX_BYTES`, if its length differs from
    `meta["file_size"]`, if `verify_sha256(body, meta["sha256"])` is False, or if
    `expected_sha256` is given and `verify_sha256(body, expected_sha256)` is False.
  - Write atomically: create `media_dir` if needed, write to the same name plus `.part`,
    then `Path.replace` onto the final path. Nothing is left at the final path on failure.
  - Return `{"path": Path, "mime_type": meta["mime_type"], "bytes": int,
    "sha256": hex digest str, "cached": bool}`.
- The token and the media URLs (they carry signed query strings) must never appear in any
  exception message.

"""Download WhatsApp voice notes and photos via the Graph API.

Usage example:
    from pathlib import Path
    import requests
    from whatsapp_media import download_media

    out = download_media(
        "2708226019592949",
        "GRAPH_TOKEN",
        Path("media"),
        expected_sha256=None,
        session=requests.Session(),
    )
    print(out["path"])
"""

import base64
import hashlib
import hmac
from pathlib import Path
from typing import Any, Optional

import requests

MAX_BYTES: int = 16 * 1024 * 1024

EXTENSIONS: dict[str, str] = {
    "audio/ogg": ".ogg",
    "audio/mpeg": ".mp3",
    "audio/mp4": ".m4a",
    "audio/aac": ".aac",
    "audio/amr": ".amr",
    "image/jpeg": ".jpg",
    "image/png": ".png",
}


class MediaError(Exception):
    """Raised when WhatsApp media cannot be downloaded or verified."""


def verify_sha256(data: bytes, expected: str) -> bool:
    """Return True if expected matches the SHA-256 of data.

    Accepts 64 hex digits (any case) or standard base64 of the digest.
    Returns False for anything else and never raises.
    """
    try:
        if not isinstance(data, (bytes, bytearray)):
            return False
        if not isinstance(expected, str):
            return False
        if expected == "":
            return False
        digest = hashlib.sha256(bytes(data)).digest()
        hex_digest = hashlib.sha256(bytes(data)).hexdigest()
        b64_digest = base64.b64encode(digest).decode("ascii")
        if len(expected) == 64:
            try:
                return hmac.compare_digest(expected.lower(), hex_digest)
            except Exception:
                return False
        if len(expected) == 44:
            try:
                return hmac.compare_digest(expected, b64_digest)
            except Exception:
                return False
        return False
    except Exception:
        return False


def _base_mime(mime_type: str) -> str:
    return mime_type.split(";")[0].strip().lower()


def media_path(media_dir: Path, media_id: str, mime_type: str) -> Path:
    """Return the local path for a media id and mime type."""
    if (
        not isinstance(media_id, str)
        or not media_id
        or not media_id.isascii()
        or not media_id.isdigit()
    ):
        raise MediaError("invalid media id")
    if not isinstance(mime_type, str):
        raise MediaError("unsupported media type")
    base = _base_mime(mime_type)
    ext = EXTENSIONS.get(base)
    if ext is None:
        raise MediaError("unsupported media type")
    directory = Path(media_dir)
    return directory / f"{media_id}{ext}"


def _auth_headers(token: str) -> dict[str, str]:
    return {"Authorization": f"Bearer {token}"}


def _fetch_meta(
    session: Any, meta_url: str, token: str
) -> tuple[Any, dict[str, Any]]:
    try:
        resp = session.get(
            meta_url, headers={"Authorization": f"Bearer {token}"}, timeout=30
        )
    except MediaError:
        raise
    except Exception:
        raise MediaError("metadata request failed")
    if resp is None or getattr(resp, "status_code", None) != 200:
        code = getattr(resp, "status_code", "unknown")
        raise MediaError(f"metadata request failed with status {code}")
    try:
        meta = resp.json()
    except Exception:
        raise MediaError("invalid metadata response")
    if not isinstance(meta, dict):
        raise MediaError("invalid metadata response")
    return resp, meta


def _checked_meta(meta: Any) -> tuple[str, str, str, int]:
    try:
        file_url = meta["url"]
        mime_type = meta["mime_type"]
        sha = meta["sha256"]
        size = meta["file_size"]
    except Exception:
        raise MediaError("invalid metadata response")
    if not isinstance(file_url, str) or not file_url:
        raise MediaError("invalid metadata response")
    if not isinstance(mime_type, str) or not mime_type:
        raise MediaError("invalid metadata response")
    if not isinstance(sha, str) or not sha:
        raise MediaError("invalid metadata response")
    if not isinstance(size, int) or isinstance(size, bool):
        raise MediaError("invalid metadata response")
    return file_url, mime_type, sha, size


def _fetch_bytes(session: Any, url: str, token: str) -> Any:
    try:
        return session.get(
            url, headers={"Authorization": f"Bearer {token}"}, timeout=30
        )
    except MediaError:
        raise
    except Exception:
        raise MediaError("download request failed")


def download_media(
    media_id: str,
    token: str,
    media_dir: Path,
    expected_sha256: Optional[str] = None,
    session: Any = None,
    graph_url: str = "https://graph.facebook.com",
) -> dict:
    """Download WhatsApp media and verify it, returning a result dict."""
    if (
        not isinstance(media_id, str)
        or not media_id
        or not media_id.isascii()
        or not media_id.isdigit()
    ):
        raise MediaError("invalid media id")
    if session is None:
        session = requests.Session()
    base = graph_url.rstrip("/") if isinstance(graph_url, str) else ""
    meta_url = f"{base}/{media_id}"

    _, meta = _fetch_meta(session, meta_url, token)
    file_url, mime_type, sha, file_size = _checked_meta(meta)
    path = media_path(media_dir, media_id, mime_type)

    try:
        if path.is_file():
            try:
                existing = path.read_bytes()
            except Exception:
                existing = None
            if existing is not None and verify_sha256(existing, sha):
                hex_digest = hashlib.sha256(existing).hexdigest()
                return {
                    "path": path,
                    "mime_type": mime_type,
                    "bytes": len(existing),
                    "sha256": hex_digest,
                    "cached": True,
                }
    except MediaError:
        raise
    except Exception:
        pass

    resp = _fetch_bytes(session, file_url, token)
    if getattr(resp, "status_code", None) == 404:
        _, meta = _fetch_meta(session, meta_url, token)
        file_url, mime_type, sha, file_size = _checked_meta(meta)
        path = media_path(media_dir, media_id, mime_type)
        try:
            if path.is_file():
                try:
                    existing = path.read_bytes()
                except Exception:
                    existing = None
                if existing is not None and verify_sha256(existing, sha):
                    hex_digest = hashlib.sha256(existing).hexdigest()
                    return {
                        "path": path,
                        "mime_type": mime_type,
                        "bytes": len(existing),
                        "sha256": hex_digest,
                        "cached": True,
                    }
        except MediaError:
            raise
        except Exception:
            pass
        resp = _fetch_bytes(session, file_url, token)
        if getattr(resp, "status_code", None) != 200:
            code = getattr(resp, "status_code", "unknown")
            raise MediaError(f"download failed with status {code}")
    elif getattr(resp, "status_code", None) != 200:
        code = getattr(resp, "status_code", "unknown")
        raise MediaError(f"download failed with status {code}")

    try:
        body = resp.content
    except Exception:
        raise MediaError("invalid download body")
    if not isinstance(body, (bytes, bytearray)):
        raise MediaError("invalid download body")
    body = bytes(body)

    if len(body) > MAX_BYTES:
        raise MediaError("media exceeds size limit")
    if len(body) != file_size:
        raise MediaError("size mismatch")
    if not verify_sha256(body, sha):
        raise MediaError("checksum mismatch")
    if expected_sha256 is not None and not verify_sha256(body, expected_sha256):
        raise MediaError("checksum mismatch")

    directory = Path(media_dir)
    tmp_path = directory / (path.name + ".part")
    existed_before = False
    try:
        existed_before = path.is_file()
    except Exception:
        existed_before = False
    try:
        directory.mkdir(parents=True, exist_ok=True)
        tmp_path.write_bytes(body)
        tmp_path.replace(path)
    except MediaError:
        raise
    except Exception:
        try:
            if tmp_path.is_file():
                tmp_path.unlink()
        except Exception:
            pass
        try:
            if not existed_before and path.is_file():
                path.unlink()
        except Exception:
            pass
        raise MediaError("failed to store media")
    try:
        if tmp_path.is_file():
            tmp_path.unlink()
    except Exception:
        pass

    hex_digest = hashlib.sha256(body).hexdigest()
    return {
        "path": path,
        "mime_type": mime_type,
        "bytes": len(body),
        "sha256": hex_digest,
        "cached": False,
    }
